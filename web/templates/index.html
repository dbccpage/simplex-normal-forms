2. Relational Schema Normalization & 3NF Synthesis

from normal_forms.domain.relational import FunctionalDependency, RelationalSchema, check_3nf, chase_join_dependency
from normal_forms.domain.relational_advanced import synthesize_3nf_bernstein

# 1. Declare attributes and functional dependencies
schema = RelationalSchema(
    attributes=frozenset({"order_id", "cust_id", "zip", "amount"}),
    dependencies=frozenset({
        FunctionalDependency.from_str("order_id", "cust_id,amount"),
        FunctionalDependency.from_str("cust_id", "zip")
    })
)

# 2. Synthesize 3NF subschemas (Bernstein 1976)
tables = synthesize_3nf_bernstein(schema)
print("3NF Decomposed Tables:", [sorted(t) for t in tables])

# 3. Verify lossless join via Relational Tableau Chase
chase = chase_join_dependency(schema.attributes, schema.dependencies, tables)
assert chase.is_implied is True, "Lossless join verified!"

3. Lowering Normal-Form Contracts to SQLite Triggers

import sqlite3
from normal_forms.domain.relational import FunctionalDependency
from normal_forms.ir_kernel import lower_fd_to_kernel_spec
from normal_forms.ir_lowering import compile_to_sqlite_triggers, install_and_verify_sqlite_bundle

# 1. Lower BCNF contract to SQLite triggers
spec = lower_fd_to_kernel_spec(
    fd=FunctionalDependency.from_str("dept_id", "manager_id"),
    key_field="employee_id",
    heading=("employee_id", "dept_id", "manager_id", "office"),
    contract_id="nf.classical.bcnf"
)
bundle = compile_to_sqlite_triggers(spec, table_name="employees")

# 2. Install and enforce in SQLite database
conn = sqlite3.connect(":memory:")
ok, err = install_and_verify_sqlite_bundle(conn, bundle, rows=[
    {"employee_id": "1", "dept_id": "D1", "manager_id": "M1", "office": "101"},
    {"employee_id": "2", "dept_id": "D1", "manager_id": "M2", "office": "102"} # Conflicts on dept_id -> manager_id!
])
assert ok is False, "Database trigger blocked invalid insert!"
